def matrix_chain_order(p):

    n = len(p) - 1

    # Create DP table
    dp = [[0 for _ in range(n + 1)]
          for _ in range(n + 1)]

    # Chain length
    for length in range(2, n + 1):

        for i in range(1, n - length + 2):

            j = i + length - 1

            dp[i][j] = float('inf')

            # Try every possible split
            for k in range(i, j):

                cost = (
                    dp[i][k]
                    + dp[k + 1][j]
                    + p[i - 1] * p[k] * p[j]
                )

                if cost < dp[i][j]:
                    dp[i][j] = cost

    return dp[1][n]


# Main Program

# A1 = 10 x 20
# A2 = 20 x 30
# A3 = 30 x 40

p = [10, 20, 30, 40]

result = matrix_chain_order(p)

print("Minimum number of scalar multiplications:", result)
