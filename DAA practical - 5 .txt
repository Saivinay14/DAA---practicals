def knapsack(weights, values, capacity):

    n = len(weights)

    # Create DP table
    dp = [[0 for _ in range(capacity + 1)]
          for _ in range(n + 1)]

    # Fill the table
    for i in range(1, n + 1):

        for w in range(1, capacity + 1):

            # Current item weight
            if weights[i - 1] <= w:

                # Don't take OR take the item
                dp[i][w] = max(
                    dp[i - 1][w],
                    values[i - 1] + dp[i - 1][w - weights[i - 1]]
                )

            else:

                # Item is too heavy
                dp[i][w] = dp[i - 1][w]

    return dp[n][capacity]


# Main Program
weights = [2, 1, 3, 2]
values = [12, 10, 20, 15]

capacity = 5

result = knapsack(weights, values, capacity)

print("Maximum value:", result)
